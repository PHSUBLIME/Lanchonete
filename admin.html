function processOrderSubmission(e) { e.preventDefault(); if (appState.cart.length === 0) { alert("Adicione produtos ao carrinho antes de continuar!"); return; } const name = document.getElementById("custName").value.trim(); const phone = document.getElementById("custPhone").value.trim(); const orderType = document.querySelector('input[name="orderType"]:checked').value; const paymentMethod = document.querySelector('input[name="paymentMethod"]:checked').value; const orderNotes = document.getElementById("orderNotes").value.trim() || "Nenhuma"; let deliveryAddressText = "Retirada no Balcão"; let deliveryFeeApplied = 0; if (orderType === "delivery") { const street = document.getElementById("custStreet").value.trim(); const number = document.getElementById("custNumber").value.trim(); const neighborhood = document.getElementById("custNeighborhood").value.trim(); const complement = document.getElementById("custComplement").value.trim(); const reference = document.getElementById("custReference").value.trim(); if (!street || !number || !neighborhood) { alert("Por favor, preencha Rua, Número e Bairro para entrega."); return; } deliveryAddressText = `${street}, nº ${number} - ${neighborhood}${complement ? ` (${complement})` : ""}${reference ? ` | Ref: ${reference}` : ""}`; deliveryFeeApplied = CONFIG.store.deliveryFee; } let paymentDetails = paymentMethod; if (paymentMethod === "Dinheiro") { const needChange = document.querySelector('input[name="needChangeRadio"]:checked').value === "sim"; if (needChange) { const changeFor = document.getElementById("changeAmountInput").value.trim(); paymentDetails += ` (Troco para R$ ${changeFor || "não informado"})`; } else { paymentDetails += " (Não precisa de troco)"; } } const subtotal = appState.cart.reduce((acc, item) => acc + (item.product.price * item.quantity), 0); const grandTotal = subtotal + deliveryFeeApplied; const orderNumber = Math.floor(2000 + Math.random() * 8000); // MONTAGEM DA MENSAGEM WHATSAPP const itemsListFormatted = appState.cart.map(item => { const itemTotal = item.product.price * item.quantity; return `• ${item.quantity}x ${item.product.name} - R$ ${formatMoney(itemTotal)}${item.notes ? `\n Obs: ${item.notes}` : ""}`; }).join("\n"); const message = `🍔 *NOVO PEDIDO - ${CONFIG.store.name}* *Pedido nº:* #${orderNumber} 👤 *CLIENTE:* ${name} 📱 *TELEFONE:* ${phone} 🛒 *PEDIDO:* ${itemsListFormatted} 💰 *SUBTOTAL:* R$ ${formatMoney(subtotal)} 🛵 *ENTREGA:* ${deliveryFeeApplied > 0 ? `R$ ${formatMoney(deliveryFeeApplied)}` : "Grátis (Retirada)"} 💵 *TOTAL:* R$ ${formatMoney(grandTotal)} 📍 *ENDEREÇO:* ${deliveryAddressText} 💳 *PAGAMENTO:* ${paymentDetails} 📝 *OBSERVAÇÕES:* ${orderNotes} Obrigado pelo pedido!`; // OBJETO DO PEDIDO PARA O PAINEL DO LOJISTA const orderData = { id: orderNumber.toString(), client: name, phone: phone, items: appState.cart.map(i => `${i.quantity}x ${i.product.name}`).join(", "), total: grandTotal, payment: paymentDetails, type: orderType === 'delivery' ? 'Delivery' : 'Retirada', address: deliveryAddressText, notes: orderNotes !== "Nenhuma" ? orderNotes : "", time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }), status: "Novo" }; // 1. SALVA NO STORAGE COMPARTILHADO try { let currentOrders = JSON.parse(localStorage.getItem('phsublime_orders') || '[]'); currentOrders.unshift(orderData); localStorage.setItem('phsublime_orders', JSON.stringify(currentOrders)); // 2. DISPARA COMUNICAÇÃO IMEDIATA PARA O PAINEL ADMIN const syncChannel = new BroadcastChannel('phsublime_orders_channel'); syncChannel.postMessage({ type: 'NEW_ORDER', order: orderData }); } catch (err) { console.log("Erro ao sincronizar pedido:", err); } // Prepara WhatsApp e Modal const encodedMessage = encodeURIComponent(message); const whatsappUrl = `https://wa.me/${CONFIG.store.whatsapp}?text=${encodedMessage}`; document.getElementById("successOrderId").innerText = `#${orderNumber}`; document.getElementById("successOrderTotal").innerText = `R$ ${formatMoney(grandTotal)}`; document.getElementById("btnSuccessWhatsappDirect").href = whatsappUrl; closeCheckoutModal(); document.getElementById("orderSuccessModal").classList.remove("hidden"); // Abre WhatsApp window.open(whatsappUrl, "_blank"); // Limpa carrinho appState.cart = []; updateCartBadge(); }
